## 二、实现思想

### 2.1 核心流程

```
liver.h5ad → PCA 降维向量(50维) → L2 归一化 → FAISS 建索引(HNSW) → Web 查询 → 返回 Top-K
```

### 2.2 关键设计决策

**为什么选 FAISS？**
- 中期只需实现一种库，FAISS 是最稳妥的选择：Meta 维护、文档多、内置 HNSW/IVF 等多种索引、安装简单（`pip install faiss-cpu`）。

**为什么用 PCA 而不是原始基因表达矩阵做向量？**
- 原始基因维度高（20000+），直接影响检索速度和索引质量。PCA 降到 50 维后，既保留主要变异信息，又大幅降低计算开销。

**为什么做 L2 归一化？**
- 归一化后向量内积 = 余弦相似度。FAISS 的 `IndexFlatIP`（内积索引）可直接用于余弦检索，速度快。

### 2.3 系统架构（中期简化版）

```
┌──────────────────────┐
│   index.html (前端)   │   输入细胞编号 / K值 / 索引类型
└──────────┬───────────┘
```

### 第 2 步：data_loader.py —— 数据读取与向量化

我们的数据是肝脏单细胞数据 `liver.h5ad`，已包含 PCA 降维结果（`obsm['X_pca']`）和细胞元信息（`obs` 中含 `cell_type`、`disease`、`AgeGroup` 等）。

```python
import scanpy as sc
import numpy as np


def load_and_preprocess(h5ad_path="liver.h5ad", n_pcs=50):
    """
    读取 liver .h5ad 文件，返回 (N, n_pcs) 的 float32 归一化向量和 AnnData 对象。

    数据说明：
      - obs 中含 cell_type（细胞类型）、disease（疾病状态）、AgeGroup（年龄分组）
      - obsm['X_pca'] 已有 PCA 降维结果，直接用；没有则自动跑预处理流程
    """
    adata = sc.read_h5ad(h5ad_path)

    if 'X_pca' not in adata.obsm:
        sc.pp.normalize_total(adata, target_sum=1e4)
        sc.pp.log1p(adata)
        sc.pp.highly_variable_genes(adata, n_top_genes=2000)
        adata = adata[:, adata.var.highly_variable]
        sc.pp.scale(adata)
        sc.tl.pca(adata, n_comps=n_pcs)

    vectors = adata.obsm['X_pca'][:, :n_pcs].astype(np.float32)

    # L2 归一化，使内积等价于余弦相似度
    norms = np.linalg.norm(vectors, axis=1, keepdims=True)
    norms[norms == 0] = 1
    vectors = vectors / norms

    return vectors, adata


def get_metadata(adata, cell_idx):
    """获取单个细胞的元信息字典"""
    info = {'cell_id': str(adata.obs_names[cell_idx])}
    for col in adata.obs.columns:
        val = adata.obs.iloc[cell_idx][col]
        if hasattr(val, 'item'):
            val = val.item()
        info[col] = str(val)
    return info


def get_dataset_summary(adata):
    """返回数据集概况"""
    n_pcs = adata.obsm['X_pca'].shape[1] if 'X_pca' in adata.obsm else 0
    return {
        'n_cells': adata.n_obs,
        'n_genes': adata.n_vars,
        'n_pcs': n_pcs,
        'metadata_columns': list(adata.obs.columns),
    }
```

### 第 3 步：engine.py —— ANN 检索引擎

中期至少实现一种索引，建议选 **HNSW**（不需要训练、精度高、查询快）。

```python
import faiss
import numpy as np

class ANNEngine:
    def __init__(self):
        self.index = None
        self.vectors = None
        self.index_type = None

    def build_index(self, vectors, index_type='hnsw'):
        self.vectors = vectors
        d = vectors.shape[1]

        if index_type == 'flat':
            # 精确检索（基准对比用）
            self.index = faiss.IndexFlatIP(d)

        elif index_type == 'ivf':
            nlist = int(np.sqrt(vectors.shape[0]))
            quantizer = faiss.IndexFlatIP(d)
            self.index = faiss.IndexIVFFlat(quantizer, d, nlist)
            self.index.train(vectors)
            self.index.nprobe = 10

        elif index_type == 'hnsw':
            # 推荐：无需训练，精度高，速度快
            self.index = faiss.IndexHNSWFlat(d, 32)

        self.index.add(vectors)
        self.index_type = index_type

    def search(self, query_vector, k=10):
        query = np.array(query_vector, dtype=np.float32).reshape(1, -1)
        distances, indices = self.index.search(query, k)
        return distances[0], indices[0]

    def save_index(self, path):
        faiss.write_index(self.index, path)

    def load_index(self, vectors, path):
        self.vectors = vectors
        self.index = faiss.read_index(path)
```

### 第 4 步：app.py —— Flask 后端

```python
from flask import Flask, render_template, request, jsonify
from data_loader import load_and_preprocess, get_metadata, get_dataset_summary
from engine import ANNEngine
import numpy as np

app = Flask(__name__)
engine = ANNEngine()
adata = None
cell_ids = None

@app.before_first_request
def init():
    global adata, cell_ids
    vectors, adata = load_and_preprocess('liver.h5ad', n_pcs=50)
    cell_ids = adata.obs_names.tolist()
    engine.build_index(vectors, index_type='hnsw')

@app.route('/')
def index():
    return render_template('index.html')

@app.route('/api/search', methods=['POST'])
def search():
    data = request.json
    mode = data.get('mode', 'id')
    k = int(data.get('k', 10))
    idx_type = data.get('index_type', 'hnsw')

    if engine.index_type != idx_type:
        engine.build_index(engine.vectors, index_type=idx_type)

    if mode == 'id':
        query_vec = engine.vectors[int(data['cell_index'])]
    else:
        query_vec = np.array(data['vector'], dtype=np.float32)

    distances, indices = engine.search(query_vec, k=k)

    results = []
    for rank, (dist, idx) in enumerate(zip(distances, indices)):
        idx = int(idx)
        item = {'rank': rank + 1, **get_metadata(adata, idx),
                'similarity': float(dist)}
        results.append(item)

    return jsonify({'results': results, 'index_type': engine.index_type})

@app.route('/api/status')
def status():
    summary = get_dataset_summary(adata) if adata else {}
    return jsonify({
        **summary,
        'n_dims': engine.vectors.shape[1] if engine.vectors is not None else 0,
        'index_type': engine.index_type,
    })

if __name__ == '__main__':
    app.run(debug=True, port=5000)
```
